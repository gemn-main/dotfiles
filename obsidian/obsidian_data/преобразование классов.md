```python
import lightgbm as lgb 
from sklearn.model_selection import train_test_split 
from lightgbm import LGBMClassifier, LGBMRegressor
from  sklearn.metrics import f1_score
import seaborn as sns 

df = sns.load_dataset('titanic')
df = df.drop('alive', axis=1)

y = df['survived']
x = df.drop('survived', axis=1)

cat_features = ['sex', 'embarked', 'class','who','deck','embark_town']
for i in cat_features:
    x[i] = x[i].astype('category')


x_train, x_test, y_train, y_test = train_test_split(x, y, test_size=0.2)

model = LGBMClassifier()

model.fit(x_train, y_train, categorical_feature=cat_features)

y_pred = model.predict(x_test)

print(f1_score(y_test, y_pred))
```

`df = df.drop('alive', axis=1)` - утечка данных, убираем её (хранит ответы в буквенном формате)

`x[i] = x[i].astype('category')`  - приводим к классу те столбцы которые на надо привести к one hot encoding 

`categorical_feature=cat_features` - передаем столбцы которые преобразовали на всякий
